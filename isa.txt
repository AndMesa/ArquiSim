# =====================================================================
#  isa.txt -- El ISA de ArquiSim, un MIPS reducido (el CONTRATO entre software y hardware)
# =====================================================================
#  Este archivo define QUÉ instrucciones existen y CÓMO se codifican.
#  Ya trae add, addi y las instrucciones de las Etapas 1 a 3: usted agrega aquí cada
#  instrucción nueva (con la codificación dada en clase) y la implementa en src/ejecutar.c.
#  El simulador lo usa para ensamblar los programas (.asm -> binario) y
#  para decodificar cada instrucción.  QUÉ HACE cada una se implementa
#  en src/ejecutar.c.
#
#  Columnas:
#    mnemónico  formato  opcode(6 bits)  funct(6 bits o -)  operandos  [nombre interno]
#
#  Formatos (32 bits):
#    R:  op(6) | rs(5) | rt(5) | rd(5) | shamt(5) | funct(6)
#    I:  op(6) | rs(5) | rt(5) | inmediato(16)
#    J:  op(6) | dirección(26)
#
#  Operandos (en el orden en que se escriben en ensamblador):
#    rd, rs, rt  registros            imm     inmediato de 16 bits
#    shamt       desplazamiento 0-31  label   etiqueta de salto
#    imm(rs)     base + desplazamiento (acceso a memoria)
#    rd+rs       un registro escrito que va a DOS campos (forma de 2 direcciones)
#    rd=$v0      campo implícito: no se escribe, siempre es ese registro
#
#  Columna opcional "nombre interno": identifica la instrucción en ejecutar.c.
#  Si se omite, es igual al mnemónico.
#
#  Un mismo mnemónico puede tener varias formas, que el ensamblador distingue
#  por el número de operandos escritos:
#    - misma codificación y mismo nombre:  otra sintaxis de la misma instrucción
#        add  R  000000  100000  rd+rs,rt              # add $t1, $t2
#    - otra codificación y otro nombre:    otra instrucción con el mismo mnemónico
#        add  R  000000  110000  rt         adda       # add $t1  (ACC = ACC + $t1)
#
#  Pseudoinstrucción: un mnemónico nuevo con la codificación y el nombre de
#  una instrucción real, con operandos implícitos:
#        move R  000000  100000  rd,rs,rt=$zero  add    # move $t0, $t1  ->  add $t0, $t1, $zero
#  (li con un valor de 32 bits se expande sola a lui + ori si existen en el ISA)
# =====================================================================

# ---------------- Instrucciones del proyecto base ----------------
add    R  000000  100000  rd,rs,rt     # rd = rs + rt        (excepción si desborda)
addi   I  001000  -       rt,rs,imm    # rt = rs + imm       (imm con signo; excepción si desborda)

# ---------------- Etapa 1: aritméticas (tipo R) ----------------
addu   R  000000  100001  rd,rs,rt     # rd = rs + rt        (sin excepción)
sub    R  000000  100010  rd,rs,rt     # rd = rs - rt        (excepción si desborda)
subu   R  000000  100011  rd,rs,rt     # rd = rs - rt        (sin excepción)

# ---------------- Etapa 1: lógicas (tipo R) ----------------
and    R  000000  100100  rd,rs,rt     # rd = rs AND rt
or     R  000000  100101  rd,rs,rt     # rd = rs OR rt
xor    R  000000  100110  rd,rs,rt     # rd = rs XOR rt
nor    R  000000  100111  rd,rs,rt     # rd = NOT (rs OR rt)

# ---------------- Etapa 1: desplazamientos (tipo R) ----------------
sll    R  000000  000000  rd,rt,shamt  # rd = rt << shamt
srl    R  000000  000010  rd,rt,shamt  # rd = rt >> shamt    (rellena con 0)
sra    R  000000  000011  rd,rt,shamt  # rd = rt >> shamt    (conserva el signo)

# ---------------- Etapa 1: con inmediato (tipo I) ----------------
addiu  I  001001  -       rt,rs,imm    # rt = rs + imm       (imm con signo; sin excepción)
andi   I  001100  -       rt,rs,imm    # rt = rs AND imm     (imm extendido con ceros)
ori    I  001101  -       rt,rs,imm    # rt = rs OR imm      (imm extendido con ceros)
xori   I  001110  -       rt,rs,imm    # rt = rs XOR imm     (imm extendido con ceros)
lui    I  001111  -       rt,imm       # rt = imm << 16

# ---------------- Etapa 2: memoria (base + desplazamiento) ----------------
lw     I  100011  -       rt,imm(rs)   # rt = MEM32[rs + imm]
sw     I  101011  -       rt,imm(rs)   # MEM32[rs + imm] = rt
lb     I  100000  -       rt,imm(rs)   # rt = MEM8[rs + imm]  (extensión de signo)
lbu    I  100100  -       rt,imm(rs)   # rt = MEM8[rs + imm]  (extensión con ceros)
sb     I  101000  -       rt,imm(rs)   # MEM8[rs + imm] = 8 bits menos significativos de rt

# ---------------- Etapa 3: comparaciones ----------------
slt    R  000000  101010  rd,rs,rt     # rd = (rs < rt) ? 1 : 0   con signo
sltu   R  000000  101011  rd,rs,rt     # rd = (rs < rt) ? 1 : 0   sin signo
slti   I  001010  -       rt,rs,imm    # rt = (rs < imm) ? 1 : 0  con signo

# ---------------- Etapa 3: saltos condicionales (relativos al PC) ----------------
beq    I  000100  -       rs,rt,label  # si rs == rt: PC = PC + 4 + imm*4
bne    I  000101  -       rs,rt,label  # si rs != rt: PC = PC + 4 + imm*4

# ---------------- Etapa 3: saltos incondicionales ----------------
j      J  000010  -       label        # PC = (PC+4)[31:28] | dirección*4
jal    J  000011  -       label        # $ra = PC + 4;  PC = (PC+4)[31:28] | dirección*4
jr     R  000000  001000  rs           # PC = rs

# ---------------- Etapa 3: pseudoinstrucciones (no se implementan en ejecutar.c) ----------------
move   R  000000  100000  rd,rs,rt=$zero           add    # move $t0, $t1   ->  add  $t0, $t1, $zero
not    R  000000  100111  rd,rs,rt=$zero           nor    # not $t0, $t1    ->  nor  $t0, $t1, $zero
clear  R  000000  100000  rd,rs=$zero,rt=$zero     add    # clear $t0       ->  add  $t0, $zero, $zero
nop    R  000000  000000  -                        sll    # nop             ->  sll  $zero, $zero, 0
li     I  001000  -       rt,imm,rs=$zero          addi   # li $t0, 42      ->  addi $t0, $zero, 42
la     I  001000  -       rt,imm,rs=$zero          addi   # la $t0, A       ->  addi $t0, $zero, A
b      I  000100  -       label,rs=$zero,rt=$zero  beq    # b LOOP          ->  beq  $zero, $zero, LOOP
